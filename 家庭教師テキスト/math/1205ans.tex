    \[\frac{d}{dx}F(x)=\lim_{h\to 0}\frac{F(x+h)-F(x)}{h}=\lim_{h\to 0}\frac{1}{h}\int_x^{x+h}f(t)dt\]

\noindent \underline{{\large{\textbf{4．定積分と不等式}}}}
\begin{ans}
\begin{itemize}
    \item [(1)]$f(x)=e^x-x^2$とおく．$f'(x)=e^x-2x$，$f''(x)=e^x-2$．
    $x\geqq 1$において$e^x\geqq e > 2$より$f''(x)>0$．よって$f'(x)$は単調増加．
    $f'(1)=e-2>0$より，$x\geqq 1$で$f'(x)>0$．よって$f(x)$も単調増加．
    $f(1)=e-1>0$であるから，常に$f(x)>0$．すなわち$e^x>x^2$が成り立つ．
    \item [(2)]直接計算する．
    \[\int_0^x te^{-t}dt = [-te^{-t}]_0^x + \int_0^x e^{-t}dt = -xe^{-x} -e^{-x} + 1\]
    $x\to \infty$のとき，$xe^{-x}\to 0, e^{-x}\to 0$であるから，極限値は$\underline{\boldsymbol{1}}$．
\end{itemize}
\end{ans}

\noindent \underline{{\large{\textbf{5．区分求積法}}}}
\begin{rembox}
    \textbf{定積分と極限(区分求積法)}
    \[\boldsymbol{\lim_{n\to\infty}\frac{1}{n}\sum_{k=1}^nf\left(\frac kn\right)=\int_0^1 f(x)dx}\]
\end{rembox}
\begin{ans}
\begin{itemize}
    \item [(1)]与式$=\lim_{n\to\infty}\frac{1}{n}\sum_{k=1}^n \left(\frac{k}{n}\right)^2 e^{\frac{k}{n}}$
    \[=\int_0^1 x^2e^x dx = [x^2e^x]_0^1 - \int_0^1 2xe^x dx = e - 2[xe^x-e^x]_0^1 = e - 2(e-e+1) = \underline{\boldsymbol{e-2}}\]
     \item [(2)]与式$=\lim_{n\to\infty}\frac{1}{n}\sum_{k=1}^n \frac{2(k/n)}{1+(k/n)^2}$
     \[=\int_0^1 \frac{2x}{1+x^2}dx = [\log(1+x^2)]_0^1 = \underline{\boldsymbol{\log 2}}\]
          \item [(3)]与式$=\lim_{n\to\infty}\frac{1}{n}\sum_{k=1}^n\frac{1}{1+\frac{k}{n}}$
          \[=\int_0^1 \frac{1}{1+x}dx = [\log(1+x)]_0^1 = \underline{\boldsymbol{\log 2}}\]
\end{itemize}
\end{ans}
